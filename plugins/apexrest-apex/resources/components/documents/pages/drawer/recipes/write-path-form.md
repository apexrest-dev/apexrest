# Drawer form with a package write path

List page opening an end-side drawer that loads one row, saves only through a PL/SQL API with a ROW_VERSION lost-update guard, keeps the drawer open on business errors and refreshes the list after closing; no form-region Automatic Row DML.

Status: ready.

Component: component:pages/drawer.

## Adaptation

- Choose free page numbers and rename all P1_ and P2_ items, links and the closeDialog request list consistently.
- Rename region, button and dynamic-action identifiers before merging into an existing application.
- Preserve the target application authentication and authorization.
- Map the invokeApi processes to the project's existing write API; keep the ROW_VERSION parameter and do not add Automatic Row DML beside it.
- Keep closeDialog gated by the CREATE and APPLY-CHANGES requests so a business error leaves the drawer open.
- When adapting into an APEX 26.2 application, give each process on the drawer page a unique advanced.executionMappingIdentifier (for example page * 100 + n); the 26.2 compiler otherwise reports DUPLICATED_COMPONENT.
- Package errors are shown with their ORA-2000x prefix; register an application error-handling function if users should see only the message text.

Apply the files below as an overlay to component:scaffold for offline validation. In an existing project, adapt and merge the component into its existing page and shared objects. The example is not an import authorization.

                heading: Name

            reportColumnQueryId: 3

            }

        layout {
            sequence: 10

        }
        appearance {
            buttonTemplate: @/text

            templateOptions: #DEFAULT#
        }

            action: redirectThisApp

                page: 2
                clearCache: 2

        name: Refresh rows after the drawer closes
        execution {
            sequence: 10
        }

            event: apexafterclosedialog
            selectionType: jsExpression

            action: refresh
            affectedElements {
                selectionType: region
                region: @rows
            }
            execution {
                sequence: 10
                fireOnInit: false
            }
        )
    )
)

## prerequisites/example-rows.sql

```sql
-- Optional isolated development fixture. Never run automatically during compilation/import.
-- Apply only to an explicitly authorized project schema; adapt existing data models instead of replacing them.
create table APEXREST_EXAMPLE_ROWS (
    ID number generated by default on null as identity primary key,
    NAME varchar2(200 char) not null,
    STATUS varchar2(20 char) default 'Open' not null,
    ROW_VERSION number default 1 not null,
    constraint APEXREST_EXAMPLE_ROWS_STATUS check (STATUS in ('Open', 'Closed'))
);
create or replace trigger APEXREST_EXAMPLE_ROWS_VERSION
before update on APEXREST_EXAMPLE_ROWS
for each row
begin
    :new.ROW_VERSION := :old.ROW_VERSION + 1;
end;
/

```

## prerequisites/example-rows-api.sql

```sql
-- Optional isolated development fixture. Never run automatically during compilation/import.
-- Apply after example-rows.sql, only to an explicitly authorized project schema.
-- The package is the only write path for APEXREST_EXAMPLE_ROWS used by the drawer page.
-- It never commits; APEX commits the page submit. ROW_VERSION is incremented by the
-- example-rows.sql trigger, so the update only matches the version the page loaded.
create or replace package APEXREST_EXAMPLE_ROWS_API authid definer as
    procedure create_row(
        p_name   in  varchar2,
        p_status in  varchar2,
        p_id     out number);

    procedure update_row(
        p_id          in number,
        p_name        in varchar2,
        p_status      in varchar2,
        p_row_version in number);
end APEXREST_EXAMPLE_ROWS_API;
/
create or replace package body APEXREST_EXAMPLE_ROWS_API as
    procedure check_data(
        p_name   in varchar2,
        p_status in varchar2,
        p_id     in number default null)
    is
        l_count pls_integer;
    begin
        if trim(p_name) is null then
            raise_application_error(-20001, 'Name is required.');
        end if;
        if p_status is null or p_status not in ('Open', 'Closed') then
            raise_application_error(-20001, 'Status must be Open or Closed.');
        end if;
        select count(*)
          into l_count
          from APEXREST_EXAMPLE_ROWS
         where upper(NAME) = upper(trim(p_name))
           and (p_id is null or ID <> p_id);
        if l_count > 0 then
            raise_application_error(-20001, 'A row named ' || trim(p_name) || ' already exists.');
        end if;
    end check_data;

    procedure create_row(
        p_name   in  varchar2,
        p_status in  varchar2,
        p_id     out number)
    is
    begin
        check_data(p_name, p_status);
        insert into APEXREST_EXAMPLE_ROWS (NAME, STATUS)
        values (trim(p_name), p_status)
        returning ID into p_id;
    end create_row;

    procedure update_row(
        p_id          in number,
        p_name        in varchar2,
        p_status      in varchar2,
        p_row_version in number)
    is
    begin
        if p_id is null then
            raise_application_error(-20001, 'The row to update was not identified.');
        end if;
        check_data(p_name, p_status, p_id);
        update APEXREST_EXAMPLE_ROWS
           set NAME = trim(p_name),
               STATUS = p_status
         where ID = p_id
           and ROW_VERSION = p_row_version;
        if sql%rowcount = 0 then
            raise_application_error(-20002, 'This row was changed by someone else since you opened it. Reload and try again.');
        end if;
    end update_row;
end APEXREST_EXAMPLE_ROWS_API;
/

```
